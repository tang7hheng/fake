# ── 加载系统缓存 ──────────────────────────────────────────
load_config() {
    if [[ -f "$CONF_FILE" ]]; then
        source "$CONF_FILE" 2>/dev/null || true
        # 校验
        if [[ -z "$SYS_TAG" ]] || ! echo "$SYS_TAG" | grep -qE "^(deb|rhel|arch|suse)$"; then
            detect_system
        fi
        # 校验剪贴板工具是否还存在
        if [[ -n "$CLIP_CMD" ]] && ! command -v "$CLIP_CMD" &>/dev/null; then
            detect_clipboard
            save_config
        fi
    else
        detect_system
        detect_clipboard
        save_config
    fi
}

detect_system() {
    if [[ -f /etc/os-release ]]; then
        . /etc/os-release
        local id="${ID:-unknown}"
        local vid="${VERSION_ID:-0}"
        SYS_VERSION_ID="$vid"
        case "$id" in
            ubuntu|debian|linuxmint|pop!_os) SYS_TAG="deb" ;;
            centos|rhel|rocky|almalinux|fedora) SYS_TAG="rhel" ;;
            arch|manjaro|endeavouros) SYS_TAG="arch" ;;
            opensuse-leap|opensuse-tumbleweed|sles) SYS_TAG="suse" ;;
            *) SYS_TAG="unknown" ;;
        esac
    else
        SYS_TAG="unknown"
        SYS_VERSION_ID="0"
    fi
}

detect_clipboard() {
    for tool in xclip xsel wl-copy pbcopy; do
        if command -v "$tool" &>/dev/null; then
            CLIP_CMD="$tool"
            case "$tool" in
                xclip) CLIP_ARGS="-selection clipboard" ;;
                xsel) CLIP_ARGS="--clipboard --input" ;;
                wl-copy) CLIP_ARGS="" ;;
                pbcopy) CLIP_ARGS="" ;;
            esac
            return
        fi
    done
    CLIP_CMD=""
    CLIP_ARGS=""
}

save_config() {
    cat > "$CONF_FILE" <<EOF
SYS_TAG=$SYS_TAG
SYS_VERSION_ID=$SYS_VERSION_ID
CLIP_CMD=$CLIP_CMD
CLIP_ARGS=$CLIP_ARGS
EOF
}

# ── 剪贴板复制 ────────────────────────────────────────────
copy_to_clipboard() {
    local text="$1"
    if [[ -n "$CLIP_CMD" ]]; then
        if [[ -n "$CLIP_ARGS" ]]; then
            echo -n "$text" | "$CLIP_CMD" $CLIP_ARGS 2>/dev/null && return 0
        else
            echo -n "$text" | "$CLIP_CMD" 2>/dev/null && return 0
        fi
    fi
    return 1
}

# ── 历史记录 ──────────────────────────────────────────────
add_history() {
    local cmd="$1" desc="$2" src="$3"
    local entry="${cmd}|${desc}|${src}"
    # 去重：如果已存在则移到最前
    local tmp=""
    local found=0
    if [[ -f "$HISTORY_FILE" ]]; then
        while IFS= read -r line; do
            if [[ "$line" == "${cmd}|"* ]]; then
                found=1
                continue
            fi
            tmp="${tmp}${line}"$'\n'
        done < "$HISTORY_FILE"
    fi
    # 写入：新记录在最前
    echo "$entry" > "$HISTORY_FILE"
    if [[ -n "$tmp" ]]; then
        echo -n "$tmp" >> "$HISTORY_FILE"
    fi
    # 只保留最近 MAX_HISTORY 条
    local count
    count=$(wc -l < "$HISTORY_FILE")
    if (( count > MAX_HISTORY )); then
        tail -n "$MAX_HISTORY" "$HISTORY_FILE" > "$HISTORY_FILE.tmp"
        mv "$HISTORY_FILE.tmp" "$HISTORY_FILE"
    fi
}

load_history() {
    HISTORY=()
    if [[ -f "$HISTORY_FILE" ]]; then
        while IFS='|' read -r cmd desc src; do
            [[ -n "$cmd" ]] && HISTORY+=("$cmd|$desc")
        done < "$HISTORY_FILE"
    fi
}

# ── 解析命令数据 ──────────────────────────────────────────
# 解析模块数据为子分类数组
# SUBCATEGORIES[模块名] = "子分类1|子分类2|..."
# SUB_ITEMS[模块名|子分类名] = "命令1|说明1\n命令2|说明2\n..."

declare -A SUBCATEGORIES
declare -A SUB_ITEMS

parse_module() {
    local mod_name="$1"
    local raw_data="${MODULE_DATA[$mod_name]}"
    local current_sub=""
    local sub_list=""
    local item_list=""

    while IFS= read -r line; do
        if [[ -z "$line" ]]; then
            continue
        fi
        if [[ "$line" == "## "* ]]; then
            # 保存上一个子分类
            if [[ -n "$current_sub" ]]; then
                SUB_ITEMS["${mod_name}|${current_sub}"]="$item_list"
            fi
            current_sub="${line#### }"
            sub_list="${sub_list}${current_sub}|"
            item_list=""
        else
            item_list="${item_list}${line}"$'\n'
        fi
    done <<< "$raw_data"

    # 保存最后一个子分类
    if [[ -n "$current_sub" ]]; then
        SUB_ITEMS["${mod_name}|${current_sub}"]="$item_list"
    fi

    # 去掉末尾的 |
    sub_list="${sub_list%|}"
    SUBCATEGORIES["$mod_name"]="$sub_list"
}

parse_all_modules() {
    for mod in "${MODULE_NAMES[@]}"; do
        parse_module "$mod"
    done
}

# ── 方向键交互式选择 ──────────────────────────────────────
# 用法: interactive_select "提示文字" 数组名 [返回值变量]
# 返回: 选中的索引(0-based)，-1表示返回，-2表示退出，-3表示搜索
interactive_select() {
    local hint="$1"
    local -n items="$2"
    local -n _ret="$3"
    local cursor=0
    local search_input=""
    local filtered_indices=()
    local total=${#items[@]}

    # 初始化过滤索引
    for ((i=0; i<total; i++)); do
        filtered_indices+=($i)
    done

    local visible_lines=$(($(tput lines 2>/dev/null || echo 24) - UI_TOP - 6))
    (( visible_lines < 5 )) && visible_lines=5
    local scroll_offset=0

    while true; do
        local display_count=${#filtered_indices[@]}

        # 调整滚动
        if (( cursor < scroll_offset )); then
            scroll_offset=cursor
        fi
        if (( cursor >= scroll_offset + visible_lines )); then
            scroll_offset=$((cursor - visible_lines + 1))
        fi

        # 重绘
        echo -en "\033[${UI_TOP}H\033[J"

        for ((i=0; i<display_count && i<visible_lines; i++)); do
            local idx=${filtered_indices[$((i + scroll_offset))]}
            local item="${items[$idx]}"
            local num=$((idx + 1))

            if (( i + scroll_offset == cursor )); then
                echo -e "  ${BG_CYAN}${BLACK} ▶ ${item} ${RESET}"
            else
                echo -e "  ${GRAY}  ${item}${RESET}"
            fi
        done

        # 底部提示
        local line_num=$((UI_TOP + visible_lines + 2))
        echo -en "\033[${line_num}H"
        echo -e "  ${GRAY}↑↓ 选择  回车确认  输入关键词搜索  b 返回  q 退出${RESET}"

        # 搜索提示
        if [[ -n "$search_input" ]]; then
            echo -en "\033[$((line_num+1))H"
            echo -e "  ${YELLOW}搜索: $search_input${RESET}"
        fi

        # 读取按键
        echo -en "$CURSOR_HIDE"
        local key
        IFS= read -rsn1 key

        if [[ "$key" == $'\x1b' ]]; then
            read -rsn2 -t 0.1 rest
            case "$rest" in
                '[A'|'[H') # Up / Home
                    if (( cursor > 0 )); then
                        (( cursor-- ))
                    fi
                    ;;
                '[B'|'[F') # Down / End
                    if (( cursor < display_count - 1 )); then
                        (( cursor++ ))
                    fi
                    ;;
                '[5') # Page Up
                    (( cursor -= visible_lines ))
                    (( cursor < 0 )) && cursor=0
                    ;;
                '[6') # Page Down
                    (( cursor += visible_lines ))
                    (( cursor >= display_count )) && cursor=$((display_count - 1))
                    ;;
                *) # Esc
                    echo -en "$CURSOR_SHOW"
                    _ret=-2
                    return
                    ;;
            esac
        elif [[ "$key" == $'\n' || "$key" == $'\r' ]]; then
            echo -en "$CURSOR_SHOW"
            if (( display_count > 0 )); then
                _ret=${filtered_indices[$cursor]}
            else
                _ret=-1
            fi
            return
        elif [[ "$key" == "b" || "$key" == "B" ]]; then
            echo -en "$CURSOR_SHOW"
            _ret=-1
            return
        elif [[ "$key" == "q" || "$key" == "Q" ]]; then
            echo -en "$CURSOR_SHOW"
            _ret=-2
            return
        elif [[ "$key" == "j" ]]; then
            if (( cursor < display_count - 1 )); then
                (( cursor++ ))
            fi
        elif [[ "$key" == "k" ]]; then
            if (( cursor > 0 )); then
                (( cursor-- ))
            fi
        elif [[ "$key" == $'\x7f' || "$key" == $'\x08' ]]; then
            # Backspace
            if [[ -n "$search_input" ]]; then
                search_input="${search_input%?}"
                # 重新过滤
                filtered_indices=()
                for ((i=0; i<total; i++)); do
                    if echo "${items[$i]}" | grep -qi "$search_input"; then
                        filtered_indices+=($i)
                    fi
                done
                cursor=0
                scroll_offset=0
            fi
        elif [[ -n "$key" && ! "$key" =~ [0-9] ]]; then
            # 普通字符 -> 搜索
            search_input="${search_input}${key}"
            filtered_indices=()
            for ((i=0; i<total; i++)); do
                if echo "${items[$i]}" | grep -qi "$search_input"; then
                    filtered_indices+=($i)
                fi
            done
            cursor=0
            scroll_offset=0
        elif [[ "$key" =~ [0-9] ]]; then
            # 数字 -> 直接跳转
            search_input="${search_input}${key}"
            local num="$search_input"
            if (( num >= 1 && num <= total )); then
                echo -en "$CURSOR_SHOW"
                _ret=$((num - 1))
                return
            fi
        fi
        echo -en "$CURSOR_SHOW"
    done
}

# ── 搜索逻辑 ──────────────────────────────────────────────
search_commands() {
    local keyword="$1"
    local results=()

    # 第1级: 精确匹配模块名
    for i in "${!MODULE_NAMES[@]}"; do
        if [[ "${MODULE_NAMES[$i]}" == *"$keyword"* ]]; then
            echo "MODULE:$i"
            return
        fi
    done

    # 第2级: 匹配模块 tags
    for i in "${!MODULE_NAMES[@]}"; do
        if echo "${MODULE_TAGS[$i]}" | grep -qi "$keyword"; then
            echo "MODULE:$i"
            return
        fi
    done

    # 第3级: 搜索命令字符串
    for mod in "${MODULE_NAMES[@]}"; do
        local raw="${MODULE_DATA[$mod]}"
        while IFS= read -r line; do
            if [[ -n "$line" && "$line" != "##"* ]] && echo "$line" | grep -qi "$keyword"; then
                results+=("$mod|$line")
            fi
        done <<< "$raw"
    done

    if (( ${#results[@]} > 0 )); then
        printf '%s\n' "${results[@]}"
    else
        echo "NOT_FOUND"
    fi
}

# ── 显示命令列表并选择 ────────────────────────────────────
show_commands_and_select() {
    local mod_name="$1"
    local sub_name="$2"
    local key="${mod_name}|${sub_name}"
    local raw="${SUB_ITEMS[$key]}"

    # 解析为命令数组
    local -a cmd_items=()
    while IFS= read -r line; do
        [[ -n "$line" ]] && cmd_items+=("$line")
    done <<< "$raw"

    if (( ${#cmd_items[@]} == 0 )); then
        return 1
    fi

    # 显示
    echo ""
    print_breadcrumb "$mod_name > $sub_name"
    print_separator

    local selected
    interactive_select "选择命令" cmd_items selected

    case $selected in
        -2) # 退出
            echo ""
            exit 0
            ;;
        -1) # 返回
            return 1
            ;;
        *)
            local item="${cmd_items[$selected]}"
            local cmd="${item%%|*}"
            copy_and_notify "$cmd" "$mod_name"
            # 不退出，继续选择
            show_commands_and_select "$mod_name" "$sub_name"
            ;;
    esac
}

    local src="${2:-}"

    if copy_to_clipboard "$cmd"; then

            echo -e "  ${YELLOW}💡 粘贴后替换 $(echo "$cmd" | grep -o '{[^}]*}') 即可使用${RESET}"

        echo -e "  ${CYAN}📋 ${BOLD}${cmd}${RESET}"
        echo -e "  ${YELLOW}💡 剪贴板不可用，请手动复制以上命令${RESET}"
        echo ""
    fi

    # 记录历史
    local desc=""
    # 从原始数据中查找描述
    for mod in "${MODULE_NAMES[@]}"; do
        local raw="${MODULE_DATA[$mod]}"
        while IFS= read -r line; do
            if [[ "$line" == "${cmd}|"* ]]; then
                desc="${line#*|}"
                break 2
            fi
        done <<< "$raw"
    done
    [[ -z "$desc" ]] && desc="自定义命令"
    [[ -z "$src" ]] && src="搜索"
    add_history "$cmd" "$desc" "$src"

    # 保存到临时文件供 Shell 函数使用
    echo "$cmd" > "$FAKE_CMD_FILE"
}

# ── 显示搜索结果并选择 ──────────────────────────────────
show_search_results() {
    local keyword="$1"
    shift
    local results=("$@")

    echo ""
    echo -e "  ${BOLD}搜索: \"${keyword}\" — 找到 ${#results[@]} 条${RESET}"
    print_separator

    local -a display_items=()
    for r in "${results[@]}"; do
        local item="${r#*|}"
        display_items+=("$item")
    done

    local selected
    interactive_select "选择命令" display_items selected

    case $selected in
        -2) exit 0 ;;
        -1) return 1 ;;
        *)
            local item="${display_items[$selected]}"
            local cmd="${item%%|*}"
            local mod="${results[$selected]%%|*}"
            copy_and_notify "$cmd" "$mod"
            show_search_results "$keyword" "${results[@]}"
            ;;
    esac
}

# ── 显示子分类并选择 ──────────────────────────────────────
show_subcategories() {
    local mod_name="$1"
    local subs="${SUBCATEGORIES[$mod_name]}"
    local IFS_OLD="$IFS"
    IFS='|'
    local -a sub_list=($subs)
    IFS="$IFS_OLD"

    # 统计每个子分类的命令数
    local -a display_items=()
    for sub in "${sub_list[@]}"; do
        local key="${mod_name}|${sub}"
        local raw="${SUB_ITEMS[$key]}"
        local count=0
        while IFS= read -r line; do
            [[ -n "$line" ]] && ((count++))
        done <<< "$raw"
        display_items+=("$sub  ($count 条)")
    done

    echo ""
    print_breadcrumb "$mod_name"
    print_separator

    local selected
    interactive_select "选择子分类" display_items selected

    case $selected in
        -2) exit 0 ;;
        -1) return 1 ;;
        *)
            show_commands_and_select "$mod_name" "${sub_list[$selected]}"
            show_subcategories "$mod_name"
            ;;
    esac
}

# ── 主界面 ──────────────────────────────────────────────
show_main() {
    # 构建显示列表
    local -a display_items=()

    # 最近使用
    load_history
    if (( ${#HISTORY[@]} > 0 )); then
        display_items+=("")
        display_items+=("⏱ 最近使用:")
        for h in "${HISTORY[@]}"; do
            display_items+=("  $h")
        done
        display_items+=("")
    fi

    # 模块列表
    for i in "${!MODULE_NAMES[@]}"; do
        local mod="${MODULE_NAMES[$i]}"
        local subs="${SUBCATEGORIES[$mod]}"
        local IFS_OLD="$IFS"
        IFS='|'
        local -a sub_list=($subs)
        IFS="$IFS_OLD"
        local total=0
        for sub in "${sub_list[@]}"; do
            local key="${mod}|${sub}"
            local raw="${SUB_ITEMS[$key]}"
            while IFS= read -r line; do
                [[ -n "$line" ]] && ((total++))
            done <<< "$raw"
        done
        display_items+=("$mod  ($total 条)")
    done

    echo ""
    print_header "fake - Linux 命令快速查询工具"
    print_separator

    # 记录 UI 起始行
    UI_TOP=$(($(echo "$LINES" 2>/dev/null || tput lines 2>/dev/null || echo 24) - ${#display_items[@]} - 6))
    (( UI_TOP < 1 )) && UI_TOP=1

    local selected
    interactive_select "选择" display_items selected

    case $selected in
        -2) exit 0 ;;
        -1) return ;;
        *)
            local item="${display_items[$selected]}"
            # 判断是最近使用还是模块
            if [[ "$item" == "  "* ]]; then
                # 最近使用记录
                local cmd="${item#* }"
                cmd="${cmd%%|*}"
                copy_and_notify "$cmd" "最近使用"
                show_main
            else
                # 模块
                local mod_name="${item%% *}"
                show_subcategories "$mod_name"
                show_main
            fi
            ;;
    esac
}

# ── 快捷跳转 fake 12 4 5 ─────────────────────────────────
handle_quick_jump() {
    local args=("$@")
    local level1="${args[0]:-}"
    local level2="${args[1]:-}"
    local level3="${args[2]:-}"

    # 第1级: 模块
    if [[ "$level1" =~ ^[0-9]+$ ]] && (( level1 >= 1 && level1 <= ${#MODULE_NAMES[@]} )); then
        local mod_name="${MODULE_NAMES[$((level1-1))]}"

        # 第2级: 子分类
        if [[ -n "$level2" && "$level2" =~ ^[0-9]+$ ]]; then
            local subs="${SUBCATEGORIES[$mod_name]}"
            local IFS_OLD="$IFS"
            IFS='|'
            local -a sub_list=($subs)
            IFS="$IFS_OLD"

            if (( level2 >= 1 && level2 <= ${#sub_list[@]} )); then
                local sub_name="${sub_list[$((level2-1))]}"

                # 第3级: 命令
                if [[ -n "$level3" && "$level3" =~ ^[0-9]+$ ]]; then
                    local key="${mod_name}|${sub_name}"
                    local raw="${SUB_ITEMS[$key]}"
                    local -a cmd_items=()
                    while IFS= read -r line; do
                        [[ -n "$line" ]] && cmd_items+=("$line")
                    done <<< "$raw"

                    if (( level3 >= 1 && level3 <= ${#cmd_items[@]} )); then
                        local item="${cmd_items[$((level3-1))]}"
                        local cmd="${item%%|*}"
                        copy_and_notify "$cmd" "$mod_name"
                        exit 0
                    fi
                fi

                # 没有第3级，显示命令列表
                show_commands_and_select "$mod_name" "$sub_name"
                exit 0
            fi
        fi

        # 没有第2级，显示子分类
        show_subcategories "$mod_name"
        exit 0
    fi
}

# ── 特殊命令处理 ──────────────────────────────────────────
handle_special() {
    case "${1:-}" in
        --detect)
            detect_system
            detect_clipboard
            save_config
            echo -e "  ${GREEN}✅ 系统配置已更新${RESET}"
            echo -e "  SYS_TAG=$SYS_TAG"
            echo -e "  CLIP_CMD=$CLIP_CMD"
            exit 0
            ;;
        --inject)
            inject_shell_func
            exit 0
            ;;
        --doctor)
            run_doctor
            exit 0
            ;;
        --update)
            echo -e "  ${BOLD}正在检查更新...${RESET}"
            echo ""
            local tmp_file="/tmp/fake_update"
            local github_raw="https://raw.githubusercontent.com/tang7hheng/fake/main/fake"
            if command -v curl &>/dev/null; then
                if curl -fsSL "$github_raw" -o "$tmp_file" 2>/dev/null; then
                    sudo cp "$tmp_file" /usr/local/bin/fake 2>/dev/null && \
                    sudo chmod +x /usr/local/bin/fake 2>/dev/null
                    echo -e "  ${GREEN}✅ 已更新到最新版本${RESET}"
                    echo -e "  ${GRAY}GitHub: https://github.com/tang7hheng/fake${RESET}"
                    rm -f "$tmp_file"
                else
                    echo -e "  ${YELLOW}⚠ 下载失败，请检查网络连接${RESET}"
                    echo -e "  ${GRAY}也可以手动更新: git clone https://github.com/tang7hheng/fake.git${RESET}"
                fi
            elif command -v wget &>/dev/null; then
                if wget -qO "$tmp_file" "$github_raw" 2>/dev/null; then
                    sudo cp "$tmp_file" /usr/local/bin/fake 2>/dev/null && \
                    sudo chmod +x /usr/local/bin/fake 2>/dev/null
                    echo -e "  ${GREEN}✅ 已更新到最新版本${RESET}"
                    echo -e "  ${GRAY}GitHub: https://github.com/tang7hheng/fake${RESET}"
                    rm -f "$tmp_file"
                else
                    echo -e "  ${YELLOW}⚠ 下载失败，请检查网络连接${RESET}"
                    echo -e "  ${GRAY}也可以手动更新: git clone https://github.com/tang7hheng/fake.git${RESET}"
                fi
            else
                echo -e "  ${YELLOW}⚠ 需要 curl 或 wget 来更新${RESET}"
                echo -e "  ${GRAY}也可以手动更新: git clone https://github.com/tang7hheng/fake.git${RESET}"
            fi
            exit 0
            ;;
        -h|--help|help)
            show_help
            exit 0
            ;;
        -l|--list)
            for i in "${!MODULE_NAMES[@]}"; do
                echo "  [$((i+1))] ${MODULE_NAMES[$i]}"
            done
            exit 0
            ;;
    esac
}

inject_shell_func() {
    local target="$HOME/.bashrc"
    [[ -n "${ZSH_VERSION:-}" ]] && target="$HOME/.zshrc"

    if grep -q "_fake_init" "$target" 2>/dev/null; then
        echo -e "  ${YELLOW}⚠ Shell 函数已存在，无需重复注入${RESET}"
        return
    fi

    cat >> "$target" << 'SHELLFUNC'

# fake - 快速预填上次复制的命令
_fake_init() {
    if [[ -f /tmp/_fake_cmd ]]; then
        local cmd=$(cat /tmp/_fake_cmd 2>/dev/null)
        if [[ -n "$cmd" ]]; then
            READLINE_LINE="$cmd"
            READLINE_POINT=${#cmd}
        fi
    fi
}
bind -x '"\C-f":_fake_init' 2>/dev/null || true
SHELLFUNC

    echo -e "  ${GREEN}✅ Shell 函数已注入 $target${RESET}"
    echo -e "  ${GRAY}按 Ctrl+F 可快速预填上次复制的命令${RESET}"
}

run_doctor() {
    local ok=0
    local fail=0
    echo ""
    echo -e "  ${BOLD}══════════════════════════════════════${RESET}"
    echo -e "  ${BOLD}  fake 诊断报告${RESET}"
    echo -e "  ${BOLD}══════════════════════════════════════${RESET}"

    # 1. 程序路径
    local fake_path
    fake_path=$(command -v fake 2>/dev/null || true)
    if [[ -n "$fake_path" ]]; then
        echo -e "  ${GREEN}[✓]${RESET} 程序路径          $fake_path"
        ok=$((ok+1))
    else
        echo -e "  ${RED}[✗]${RESET} 程序路径          未找到"
        fail=$((fail+1))
    fi

    # 2. 执行权限
    if [[ -x "$fake_path" ]]; then
        echo -e "  ${GREEN}[✓]${RESET} 执行权限          正常"
        ok=$((ok+1))
    else
        echo -e "  ${RED}[✗]${RESET} 执行权限          无执行权限"
        echo -e "       ${GRAY}修复: sudo chmod +x $fake_path${RESET}"
        fail=$((fail+1))
    fi

    # 3. 系统配置
    if [[ -f "$CONF_FILE" ]]; then
        echo -e "  ${GREEN}[✓]${RESET} 系统配置          $SYS_TAG (v$SYS_VERSION_ID)"
        ok=$((ok+1))
    else
        echo -e "  ${RED}[✗]${RESET} 系统配置          缓存文件不存在"
        echo -e "       ${GRAY}修复: fake --detect${RESET}"
        fail=$((fail+1))
    fi

    # 4. 剪贴板
    if [[ -n "$CLIP_CMD" ]] && command -v "$CLIP_CMD" &>/dev/null; then
        echo -e "  ${GREEN}[✓]${RESET} 剪贴板工具        $CLIP_CMD"
        ok=$((ok+1))
    else
        echo -e "  ${YELLOW}[!]${RESET} 剪贴板工具        不可用（不影响使用）"
        echo -e "       ${GRAY}安装: sudo apt install xclip${RESET}"
        fail=$((fail+1))
    fi

    # 5. Shell 函数
    local shell_rc="$HOME/.bashrc"
    [[ -n "${ZSH_VERSION:-}" ]] && shell_rc="$HOME/.zshrc"
    if grep -q "_fake_init" "$shell_rc" 2>/dev/null; then
        echo -e "  ${GREEN}[✓]${RESET} Shell 函数        已注入 $shell_rc"
        ok=$((ok+1))
    else
        echo -e "  ${YELLOW}[!]${RESET} Shell 函数        未注入"
        echo -e "       ${GRAY}修复: fake --inject${RESET}"
        fail=$((fail+1))
    fi

    # 6. 历史文件
    if [[ -f "$HISTORY_FILE" ]]; then
        echo -e "  ${GREEN}[✓]${RESET} 历史记录          正常"
        ok=$((ok+1))
    else
        echo -e "  ${YELLOW}[!]${RESET} 历史记录          暂无（使用后会自动创建）"
        ok=$((ok+1))
    fi

    # 7. Bash 版本
    local bash_ver="${BASH_VERSION:-unknown}"
    echo -e "  ${GREEN}[✓]${RESET} Bash 版本          $bash_ver"
    ok=$((ok+1))

    echo -e "  ${BOLD}══════════════════════════════════════${RESET}"
    local total=$((ok + fail))
    if (( fail == 0 )); then
        echo -e "  ${GREEN}状态: 一切正常 ($ok/$total)${RESET}"
    else
        echo -e "  ${YELLOW}状态: 基本可用 ($ok/$total)${RESET}"
    fi
    echo -e "  ${BOLD}══════════════════════════════════════${RESET}"
    echo ""
}

# ── 自我修复 ──────────────────────────────────────────────
self_repair() {
    # 权限修复
    if [[ ! -x "${BASH_SOURCE[0]}" ]]; then
        chmod +x "${BASH_SOURCE[0]}" 2>/dev/null && \
            echo -e "  ${YELLOW}⚠ 权限异常，已自动修复${RESET}"
    fi
}

# ── 主入口 ────────────────────────────────────────────────
main() {
    self_repair
    load_config
    init_commands
    parse_all_modules

    # 处理特殊命令
    handle_special "${1:-}"

    # 处理快捷跳转（纯数字参数）
    if [[ "${1:-}" =~ ^[0-9]+$ ]]; then
        handle_quick_jump "$@"
        exit 0
    fi

    # 处理关键词搜索
    if [[ -n "${1:-}" ]]; then
        local result
        result=$(search_commands "$1")

        if [[ "$result" == "NOT_FOUND" ]]; then
            echo ""
            echo -e "  ${YELLOW}未找到与 \"$1\" 相关的命令${RESET}"
            echo -e "  ${GRAY}试试其他关键词，或直接输入 fake 浏览所有模块${RESET}"
            echo ""
            exit 0
        elif [[ "$result" == MODULE:* ]]; then
            local mod_idx="${result#MODULE:}"
            show_subcategories "${MODULE_NAMES[$mod_idx]}"
            exit 0
        else
            local -a results=()
            while IFS= read -r line; do
                [[ -n "$line" ]] && results+=("$line")
            done <<< "$result"
            show_search_results "$1" "${results[@]}"
            exit 0
        fi
    fi

    # 默认: 显示主界面
    show_main
}

main "$@"
